Arduino Servo Motor Demo

Learn how to control a servo motor with Arduino. Click the buttons to rotate the servo.

Arduino Code for Servo Motor


// Arduino Servo Example
#include <Servo.h>

Servo myServo;
int pos = 0;

void setup() {
  myServo.attach(9); // attach servo to pin 9
}

void loop() {
  for(pos = 0; pos <= 180; pos += 1) {
    myServo.write(pos);
    delay(15);
  }
  for(pos = 180; pos >= 0; pos -= 1) {
    myServo.write(pos);
    delay(15);
  }
}
